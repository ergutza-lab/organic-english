// Estado de evaluaciones por grupo (entre semana, desde el candado del 12-nov, hasta "hasta" inclusive). // 🔴 debe: día lun-jue sin evaluación (desde el 12-nov). // 🟡 mal: día con evaluación cuya lección no es la que marca el programa. // La lección del programa se cuenta desde el lunes de la semana de inicio del grupo. var REP_EVAL_DESDE = '2026-11-12'; // candado: lo anterior quedó cerrado por Coordinación function _repEstadoEvaluaciones_(gruposInfo, alumnosPorGrupo, maestrosInfo, dataReg, hasta) { var TZ = 'America/Mexico_City'; var DIAS = ['dom','lun','mar','mié','jue','vie','sáb'], MES = ['ene','feb','mar','abr','may','jun','jul','ago','sep','oct','nov','dic']; function iso(d) { return Utilities.formatDate(d, TZ, 'yyyy-MM-dd'); } function dUTC(s) { return new Date(s + 'T12:00:00Z'); } function txt(d) { return DIAS[d.getUTCDay()] + ' ' + d.getUTCDate() + ' ' + MES[d.getUTCMonth()]; } function L(p) { return 'L' + (p > 32 ? p - 32 : p); } // Lecciones registradas por grupo y fecha var reg = {}; for (var r = 1; r < dataReg.length; r++) { var g = String(dataReg[r][1] || '').trim(), f = dataReg[r][2]; if (!g || !f) continue; var k = (f instanceof Date) ? iso(f) : String(f).trim().substring(0, 10); if (k < REP_EVAL_DESDE) continue; var m = String(dataReg[r][9] || '').match(/NIVEL\s+(\S+)-Bloque\s+(\d)\s+LESSON\s+(\d+)/i); reg[g] = reg[g] || {}; reg[g][k] = reg[g][k] || {}; reg[g][k][m ? m[1].toUpperCase() + '|' + ((+m[2] - 1) * 32 + (+m[3])) : 'X|0'] = true; } var hastaISO = iso(hasta), out = []; Object.keys(gruposInfo).sort().forEach(function (idG) { var gi = gruposInfo[idG]; if (gi.estado === 'Cancelado' || gi.tipoHorario === 'personalizado' || gi.tipoHorario === 'sabado') return; if (/^PER/i.test(idG)) return; if (!gi.idMaestro || gi.idMaestro === 'SIN ASIG.' || gi.idMaestro === 'S-01') return; if (!(alumnosPorGrupo[idG] || []).length || !gi.fechaInicio) return; var ini = dUTC(iso(new Date(gi.fechaInicio))); // Lunes de la 1a semana; si inicia vie/sáb/dom (ej. cambio de nivel en viernes), arranca el lunes siguiente var wI = ini.getUTCDay(), lun0 = new Date(ini); if (wI === 5 || wI === 6 || wI === 0) lun0.setUTCDate(ini.getUTCDate() + ((8 - wI) % 7)); else lun0.setUTCDate(ini.getUTCDate() - (wI - 1)); var desde = new Date(lun0); if (desde < dUTC(REP_EVAL_DESDE)) desde = dUTC(REP_EVAL_DESDE); // solo desde el candado (12-nov) var nivelBase = (String(gi.nivel).match(/NIVEL\s+(\S+?)-/i) || [])[1]; nivelBase = nivelBase ? nivelBase.toUpperCase() : ''; var rg = reg[idG] || {}; function lecciones(k) { // posiciones del nivel actual registradas ese día return Object.keys(rg[k] || {}).filter(function (x) { return x.split('|')[0] === nivelBase; }).map(function (x) { return +x.split('|')[1]; }); } function cuenta(k) { return Object.keys(rg[k] || {}).length; } function pos(d) { var n = 0; for (var e = new Date(lun0); e <= d; e.setUTCDate(e.getUTCDate() + 1)) { var w = e.getUTCDay(); if (w >= 1 && w <= 4) n++; } return n; } var debe = [], mal = []; for (var d = new Date(desde); iso(d) <= hastaISO && d <= dUTC(hastaISO); d.setUTCDate(d.getUTCDate() + 1)) { var w = d.getUTCDay(); if (w < 1 || w > 4) continue; var k = Utilities.formatDate(d, 'UTC', 'yyyy-MM-dd'), p = pos(d); if (p > 64) continue; // ya terminó su nivel: falta el cambio de nivel, no es error del maestro if (cuenta(k)) { var ls = lecciones(k); if (ls.length && ls.indexOf(p) === -1) mal.push({ d: new Date(d), p: p }); continue; } debe.push({ d: new Date(d), p: p }); // candado: cada día lun-jue lleva su evaluación } if (debe.length || mal.length) out.push({ idGrupo: idG, nombre: gi.nombre, horario: gi.horario, idMaestro: gi.idMaestro, nombreMaestro: (maestrosInfo[gi.idMaestro] || {}).nombre || gi.idMaestro, debe: debe, mal: mal }); }); out.sort(function (a, b) { return (b.debe.length ? 1 : 0) - (a.debe.length ? 1 : 0) || (a.idGrupo < b.idGrupo ? -1 : 1); }); return out; }
